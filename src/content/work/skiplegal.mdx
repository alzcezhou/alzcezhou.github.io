  <CsInsights>

    <CsInsight title="Show your work">Trust in AI depends on being able to verify it.</CsInsight>
    <CsInsight title="Prove it's safe">Security must be visible before sharing data.</CsInsight>
  </CsInsights>
</CsSection>

  <CsDecisions>
    <CsDecision
      title="One question per screen"
      problem="jargon at the trust-deciding moment."
      decision="isolated questions + inline definitions."
      outcome="users stay in flow."
    />
    <CsDecision
      title="Autofill or Manual"
      problem="testers felt out of control."
      decision="explicit mode toggle; AI explains itself first."
      outcome="<strong>+37% trust rating.</strong>"
    />
  </CsDecisions>
</CsSection>
